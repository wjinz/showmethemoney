## Step 1: DB 정규화 및 정보 격리

### 설계 의도
`tx_YYYY` JSON 배열을 관계형 `transactions` 테이블로 분리해 row-level 쿼리 가능하게 만들고,  
`is_private` 컬럼으로 개인 지출을 RLS로 원천 차단한다.  
`SECURITY DEFINER` RPC는 RLS를 우회하므로, **RPC 내부에서 반드시 `p_household_id`가 현재 요청자의 JWT 클레임과 일치하는지 검증**해 보안 누수를 방지한다.
<!-- Antigravity: 디자인 리뉴얼에 따라 transactions 테이블에 'empathy_score' 또는 'tag' 컬럼을 추가하여 추후 AI 넛지 위젯에서 감정 소모 피드백을 제공하는 기초 자료로 활용할 수 있습니다. -->


### 1-1. Supabase SQL — 테이블 & RLS

```sql
-- ① 거래 내역 테이블
CREATE TABLE transactions (
  id           BIGSERIAL PRIMARY KEY,
  household_id TEXT        NOT NULL,          -- 가계부 UUID (기존 household_data.id와 동일)
  date         DATE        NOT NULL,
  amount       INTEGER     NOT NULL,          -- 원 단위 (소수점 없음)
  cat          TEXT        NOT NULL DEFAULT 'etc',
  memo         TEXT        NOT NULL DEFAULT '',
  who          TEXT        NOT NULL CHECK (who IN ('husband','wife')),
  pay_method   TEXT        NOT NULL DEFAULT 'credit',
  card_id      TEXT,
  is_private   BOOLEAN     NOT NULL DEFAULT false,  -- true = 상대방 블라인드
  tx_type      TEXT        NOT NULL DEFAULT 'expense',
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_tx_household_date ON transactions (household_id, date DESC);
CREATE INDEX idx_tx_household_who  ON transactions (household_id, who);

-- ② RLS 활성화
ALTER TABLE transactions ENABLE ROW LEVEL SECURITY;

-- ③ 정책 A: 자신의 가계부만 읽기 + is_private=false 또는 본인 작성분만 노출
--    (JWT claim 'app_role' = 'husband'|'wife' 로 식별)
CREATE POLICY "tx_select" ON transactions FOR SELECT
  USING (
    household_id = current_setting('request.jwt.claims', true)::jsonb->>'household_id'
    AND (
      is_private = false
      OR who = current_setting('request.jwt.claims', true)::jsonb->>'role'
    )
  );

-- ④ 정책 B: 본인 행만 INSERT/UPDATE/DELETE
CREATE POLICY "tx_insert" ON transactions FOR INSERT
  WITH CHECK (
    household_id = current_setting('request.jwt.claims', true)::jsonb->>'household_id'
    AND who = current_setting('request.jwt.claims', true)::jsonb->>'role'
  );

CREATE POLICY "tx_modify" ON transactions FOR UPDATE USING (
  who = current_setting('request.jwt.claims', true)::jsonb->>'role'
);

CREATE POLICY "tx_delete" ON transactions FOR DELETE USING (
  who = current_setting('request.jwt.claims', true)::jsonb->>'role'
);

-- ⑤ SOS 요청 테이블 (Step 4용)
CREATE TABLE sos_requests (
  id           BIGSERIAL PRIMARY KEY,
  household_id TEXT        NOT NULL,
  requester    TEXT        NOT NULL CHECK (requester IN ('husband','wife')),
  amount       INTEGER     NOT NULL,
  reason       TEXT        NOT NULL,
  repay_plan   TEXT        NOT NULL DEFAULT '다음달 차감',
  status       TEXT        NOT NULL DEFAULT 'pending'
                           CHECK (status IN ('pending','approved','rejected')),
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  resolved_at  TIMESTAMPTZ
);

ALTER TABLE sos_requests ENABLE ROW LEVEL SECURITY;
CREATE POLICY "sos_all" ON sos_requests FOR ALL
  USING (household_id = current_setting('request.jwt.claims', true)::jsonb->>'household_id');
```

### 1-2. SECURITY DEFINER RPC — 월별 총지출 집계

> **보안 원칙**: SECURITY DEFINER는 RLS를 우회하므로, 함수 내부에서 호출자의 JWT `household_id` 클레임과 `p_household_id` 파라미터가 일치하는지 명시적으로 검증한다. 불일치 시 빈 결과 반환으로 조용히 차단 (에러 메시지로 구조 노출 방지).

```sql
-- 상대방 private 내역 포함한 월합산 반환 (대시보드 차트용)
CREATE OR REPLACE FUNCTION get_monthly_total_expenses(
  p_household_id TEXT,
  p_year         INT,
  p_month        INT
)
RETURNS TABLE (who TEXT, total BIGINT)
LANGUAGE plpgsql
SECURITY DEFINER   -- RLS 우회, 함수 소유자 권한으로 실행
STABLE
AS $$
DECLARE
  v_jwt_household_id TEXT;
BEGIN
  -- ★ JWT 클레임에서 household_id 추출하여 파라미터와 대조
  v_jwt_household_id := current_setting('request.jwt.claims', true)::jsonb->>'household_id';

  -- 불일치 시 빈 결과 반환 (403 에러 대신 조용한 차단)
  IF v_jwt_household_id IS DISTINCT FROM p_household_id THEN
    RETURN;
  END IF;

  RETURN QUERY
    SELECT t.who, SUM(t.amount)::BIGINT AS total
    FROM transactions t
    WHERE t.household_id = p_household_id
      AND EXTRACT(YEAR  FROM t.date) = p_year
      AND EXTRACT(MONTH FROM t.date) = p_month
      AND t.tx_type = 'expense'
    GROUP BY t.who;
END;
$$;

-- 호출 예시 (클라이언트)
-- const { data } = await supabase.rpc('get_monthly_total_expenses',
--   { p_household_id: hid, p_year: 2026, p_month: 4 });
```

### 1-3. JWT Custom Claims 설정 & 마이그레이션 전략

Supabase Auth가 없으므로 **anon key + RPC/Service Role** 방식을 사용한다.  
App이 로그인 없이 `householdId + myRole`만 사용 중이므로 단기적으로는  
`service_role` 키를 Vercel 함수에서만 사용하고, 클라이언트는 `anon` 키로 RPC만 호출한다.

> **이행 전략**: 기존 `household_data` + `tx_YYYY` 구조를 유지하면서  
> 신규 `transactions` 테이블을 병행 사용 → 마이그레이션 완료 후 구 구조 제거.

#### 마이그레이션 중복 방지 — `migrated_to_rdb` 플래그

`App.jsx` 초기 구동 시 자동 마이그레이션을 실행하되, **`household_data`에 `key='migrated_to_rdb'`** 플래그를 저장해 중복 실행을 방지한다.

```js
// src/App.jsx — loadShared 내부, 기존 tx_YYYY 로드 이후에 추가
const migrateToRdb = async (hid, allData) => {
  // 이미 마이그레이션 완료된 경우 skip
  if (allData['migrated_to_rdb'] === true) return;

  // tx_YYYY 키에서 모든 연도 데이터 추출
  const txEntries = Object.entries(allData)
    .filter(([k]) => /^tx_\d{4}$/.test(k))
    .flatMap(([, rows]) => rows);

  if (txEntries.length > 0) {
    // bulk insert (Step 1-4 참조)
    await db.insertTxBatch(hid, txEntries);
  }

  // 완료 플래그 저장 (재실행 방지)
  await db.save(hid, 'migrated_to_rdb', true);
  console.log('[migrate] tx → transactions 완료:', txEntries.length, '건');
};
```

### 1-4. `src/utils/supabase.js` 변경 사항

```js
// 기존 saveTx/loadTx 아래에 추가
export const db = {
  // ...existing...

  /** @param {string} hid @param {number} y @param {number} m */
  async getMonthlyTotals(hid, y, m) {
    const { data, error } = await supabase.rpc('get_monthly_total_expenses', {
      p_household_id: hid, p_year: y, p_month: m
    });
    if (error) throw error;
    // [{who:'husband',total:1200000},{who:'wife',total:800000}]
    return /** @type {{who:string,total:number}[]} */ (data ?? []);
  },

  /**
   * 단건 insert (EntryView 등 단일 입력용)
   * @param {string} hid
   * @param {import('../constants').TxItem & {is_private?:boolean}} tx
   */
  async insertTx(hid, tx) {
    const { error } = await supabase.from('transactions').insert({
      household_id: hid,
      date:        tx.date,
      amount:      tx.amount,
      cat:         tx.cat,
      memo:        tx.memo,
      who:         tx.who,
      pay_method:  tx.payMethod,
      card_id:     tx.cardId ?? null,
      is_private:  tx.is_private ?? false,
      tx_type:     tx.type ?? 'expense',
    });
    if (error) throw error;
  },

  /**
   * 다건 bulk insert — CardScanSheet, 마이그레이션 등 다중 저장 시 사용
   * 단건 반복 대비 Supabase 왕복 횟수를 1회로 줄임
   * @param {string} hid
   * @param {Array<import('../constants').TxItem & {is_private?:boolean}>} txList
   */
  async insertTxBatch(hid, txList) {
    if (!txList.length) return;
    const rows = txList.map(tx => ({
      household_id: hid,
      date:        tx.date,
      amount:      tx.amount,
      cat:         tx.cat,
      memo:        tx.memo,
      who:         tx.who,
      pay_method:  tx.payMethod,
      card_id:     tx.cardId ?? null,
      is_private:  tx.is_private ?? false,
      tx_type:     tx.type ?? 'expense',
    }));
    const { error } = await supabase.from('transactions').insert(rows);
    if (error) throw error;
  },

  /** @param {string} hid @param {number} id */
  async deleteTxRow(hid, id) {
    const { error } = await supabase.from('transactions')
      .delete().eq('household_id', hid).eq('id', id);
    if (error) throw error;
  },
};
```

### 2-2. `public/sw.js` — Share Target + Cache-First 오프라인 캐싱

```js
// public/sw.js
const CACHE_NAME = 'budget-v1';

// 선캐싱 대상 — Vite 빌드 후 실제 해시 파일명으로 교체 필요 (또는 workbox 사용 시 자동화)
const PRECACHE_URLS = ['/', '/index.html'];

// ① 설치 시 핵심 에셋 선캐싱 (Cache-First의 전제)
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then(cache => cache.addAll(PRECACHE_URLS))
  );
  self.skipWaiting();
});

// ② 구버전 캐시 정리
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then(keys =>
      Promise.all(keys.filter(k => k !== CACHE_NAME).map(k => caches.delete(k)))
    )
  );
  self.clients.claim();
});

// ③ Cache-First 전략 — 캐시 히트 시 즉시 반환, miss 시 네트워크 후 캐시 저장
self.addEventListener('fetch', (event) => {
  const url = new URL(event.request.url);

  // Share Target POST 처리 (별도 분기)
  if (url.pathname === '/share-handler' && event.request.method === 'POST') {
    event.respondWith((async () => {
      const data = await event.request.formData();
      const image = data.get('image');   // File | null
      const text  = data.get('text');    // 문자 내용 (카드 알림 텍스트)

      const clients = await self.clients.matchAll({ type: 'window' });
      for (const client of clients) {
        if (image instanceof File) {
          const buf = await image.arrayBuffer();
          client.postMessage({ type: 'SHARE_IMAGE', buf, name: image.name });
        } else if (text) {
          client.postMessage({ type: 'SHARE_TEXT', text });
        }
      }
      return Response.redirect('/', 303);
    })());
    return;
  }

  // GET 요청만 캐싱 (API 호출, Supabase 요청은 제외)
  if (event.request.method !== 'GET') return;
  if (url.hostname.includes('supabase') || url.pathname.startsWith('/api/')) return;

  event.respondWith(
    caches.match(event.request).then(cached => {
      if (cached) return cached;
      return fetch(event.request).then(response => {
        // 유효한 응답만 캐시 저장
        if (response.ok && response.type === 'basic') {
          const clone = response.clone();
          caches.open(CACHE_NAME).then(cache => cache.put(event.request, clone));
        }
        return response;
      });
    })
  );
});
```

### 2-4. `src/components/ReceiptSplitter.jsx` — OCR 다중 항목 분리 UI (Coral 형광펜)

```jsx

/**
 * ReceiptSplitter — OCR 결과 N개 항목을 공동/개인으로 분류
 * @param {{ items: ScannedItem[], onConfirm: (shared: ScannedItem[], private: ScannedItem[]) => void, onClose: () => void }} props
 *
 * @typedef {{ id:string, date:string, amount:number, cat:string, memo:string }} ScannedItem
```

## Step 4: SOS 예산 청구 시스템

### 설계 의도
용돈 90% 소진 시 SOS 버튼 활성화, 요청은 `sos_requests` 테이블에 저장되고  
파트너 기기에 Realtime으로 전달된다. 상태 머신은 `pending→approved|rejected` 2단계.  
**SOS가 승인(approved)되는 시점에 해당 금액을 `transactions` 테이블에 자동 insert해 데이터 정합성을 유지한다.** SQL 트리거 또는 클라이언트 `resolveSos` 핸들러 중 어느 쪽에서 처리할지 결정 필요.

> **권장: SQL 트리거 방식** — 클라이언트 버그로 인한 누락 가능성을 원천 차단

### 4-0. SOS 승인 시 자동 트랜잭션 insert — SQL 트리거

```sql
-- SOS 승인 시 transactions 테이블에 자동 insert
CREATE OR REPLACE FUNCTION on_sos_approved()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  -- status가 'approved'로 변경될 때만 실행
  IF NEW.status = 'approved' AND OLD.status = 'pending' THEN
    INSERT INTO transactions (
      household_id, date, amount, cat, memo,
      who, pay_method, is_private, tx_type
    ) VALUES (
      NEW.household_id,
      CURRENT_DATE,
      NEW.amount,
      'etc',                          -- 카테고리: 기타 (추후 사용자 지정 가능)
      '가불: ' || NEW.reason,         -- memo에 사유 기록
      NEW.requester,
      'cash',
      true,                           -- 개인 지출로 처리 (is_private=true)
      'expense'
    );
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER trg_sos_approved
  AFTER UPDATE ON sos_requests
  FOR EACH ROW EXECUTE FUNCTION on_sos_approved();
```

```js
/** @typedef {{ id:number, household_id:string, requester:string, amount:number,
 *              reason:string, repay_plan:string, status:string, created_at:string }} SosRequest */

/** @param {string} hid @param {Omit<SosRequest,'id'|'household_id'|'created_at'|'status'>} req */
async createSosRequest(hid, req) {
  const { data, error } = await supabase.from('sos_requests').insert({
    household_id: hid,
    requester:    req.requester,
    amount:       req.amount,
    reason:       req.reason,
    repay_plan:   req.repay_plan,
    status:       'pending',
  }).select().single();
  if (error) throw error;
  return data;
},

/**
 * SOS 승인/거절 — 트리거가 approved 시 transactions에 자동 insert
 * @param {number} id @param {'approved'|'rejected'} status
 */
async resolveSos(id, status) {
  const { error } = await supabase.from('sos_requests')
    .update({ status, resolved_at: new Date().toISOString() })
    .eq('id', id);
  if (error) throw error;
  // ★ SQL 트리거(trg_sos_approved)가 approved 시 transactions 자동 insert
},

/** @param {string} hid */
async loadPendingSos(hid) {
  const { data, error } = await supabase.from('sos_requests')
    .select('*').eq('household_id', hid)
    .eq('status', 'pending').order('created_at', { ascending: false });
  if (error) throw error;
  return /** @type {SosRequest[]} */ (data ?? []);
},
```

```jsx
  const REPAYS  = ['다음달 차감', '2개월 분할', '상여금 차감'];

  const handleSubmit = async () => {
    if (!amount || !reason) return;
    setLoading(true);
    try {
      await onSubmit({
        requester:  myRole,
        amount:     parseInt(amount.replace(/,/g, '')),
        reason,
        repay_plan: repayPlan,
      });
      onClose();
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={overlayStyle}>
      <div style={sheetStyle}>
        <div style={{fontSize:14,fontWeight:800,marginBottom:16}}>🆘 용돈 가불 신청</div>
        {/* ... 입력 폼 생략 (이전 스니팻 동일) ... */}
        <button onClick={handleSubmit} disabled={loading || !amount || !reason}
          style={{width:'100%',padding:14,borderRadius:12,border:'none',
            background: (!amount||!reason) ? 'var(--bg3)' : 'var(--red)',
            color:'#fff',fontWeight:800,fontSize:14}}>
          {loading ? '전송 중...' : '파트너에게 요청 보내기'}
        </button>
      </div>
    </div>
  );
}
```

### 5-2. Ring Gauge 컴포넌트 — SVG 원형 게이지

```jsx
/**
 * 잔여 용돈 원형 게이지 — 세부 내역 없이 % 수치만 노출
 * @param {{ pct: number }} props
 */
function RingGauge({ pct }) {
  const r = 54, circ = 2 * Math.PI * r;
  const dash = circ * (1 - pct / 100);

  return (
    <svg width={140} height={140} style={{overflow:'visible'}}>
      {/* 배경 트랙 */}
      <circle cx={70} cy={70} r={r} fill="none"
        stroke="var(--border)" strokeWidth={10} />
      {/* 진행 호 — framer-motion으로 채워지는 애니메이션 */}
      <motion.circle
        cx={70} cy={70} r={r} fill="none"
        stroke="var(--accent)" strokeWidth={10}
        strokeLinecap="round"
        strokeDasharray={circ}
        initial={{ strokeDashoffset: circ }}
        animate={{ strokeDashoffset: dash }}
        transition={{ duration: 1.2, ease: 'easeOut' }}
        transform="rotate(-90 70 70)"
      />
      {/* 중앙 % 텍스트 */}
      <text x={70} y={75} textAnchor="middle"
        fontSize={22} fontWeight={900} fill="var(--text1)">
        {pct}%
      </text>
    </svg>
  );
}
```
