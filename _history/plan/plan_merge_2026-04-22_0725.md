## 2. 글로벌 헬퍼 함수

현재 `showmethemoney.html`의 `fmt()` 하나만 있는 HELPERS 섹션을 다음으로 교체한다.

// ─── HELPERS ────────────────────────────────────────────────────────────────
const ls    = (k)    => { try { return localStorage.getItem(k); } catch { return null; } };
const lsSet = (k, v) => { try { localStorage.setItem(k, v); } catch {} };
const lsGet = (k, fallback=null) => {
  const raw = ls(k);
  if (!raw) return fallback;
  try { return JSON.parse(raw); } catch { return fallback; }
};
const lsSave = (k, v) => lsSet(k, JSON.stringify(v));

const fmt = (n) => n == null ? '₩0' : '₩' + Math.abs(n).toLocaleString('ko-KR');

const toDateKey = (d = new Date()) => {
  const dt = d instanceof Date ? d : new Date(d);
  return `${dt.getFullYear()}-${String(dt.getMonth()+1).padStart(2,'0')}-${String(dt.getDate()).padStart(2,'0')}`;
};

const displayDate = (dateKey) => {
  const today = toDateKey();
  const yd = new Date(); yd.setDate(yd.getDate()-1);
  const yesterday = toDateKey(yd);
  if (dateKey === today) return '오늘';
  if (dateKey === yesterday) return '어제';
  const [,m,d] = dateKey.split('-');
  return `${parseInt(m)}월 ${parseInt(d)}일`;
};

const getCurrentMonthTx = (txList) => {
  const prefix = toDateKey().slice(0, 7); // 'YYYY-MM'
  return txList.filter(t => t.date && t.date.startsWith(prefix));
};

const genId = () => Date.now() * 1000 + (Math.random() * 1000 | 0);

const generateInviteCode = () => {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  return Array.from({length:6}, () => chars[Math.floor(Math.random()*chars.length)]).join('');
};

const loadOrGenInviteCode = () => {
  const saved = ls(LS.INVITE_CODE);
  if (saved) return saved;
  const code = generateInviteCode();
  lsSet(LS.INVITE_CODE, code);

};

### 3-2. db 객체 (src/utils/supabase.js → 인라인)
Supabase URL/KEY는 빌드 없이 HTML에 하드코딩하거나 사용자가 입력(onboarding)하도록 처리.
실제 키는 OnboardingView에서 사용자가 직접 입력 or 환경 변수 메타태그로 주입한다.

```js
// ─── SUPABASE CLIENT ────────────────────────────────────────────────────────
// Vercel 배포 시 환경변수로 주입됨 (빌드 없이 meta 태그로 전달)
const SUPABASE_URL = document.querySelector('meta[name="supabase-url"]')?.content || '';
const SUPABASE_KEY = document.querySelector('meta[name="supabase-key"]')?.content || '';
const isSupabaseReady = !!(SUPABASE_URL && SUPABASE_KEY);

const supabase = isSupabaseReady
  ? window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY)
  : null;

const db = {
  _channels: new Map(),

  async loadAll(hid) {
    if (!supabase) return {};
    const { data, error } = await supabase
      .from('household_data')
      .select('key, value')
      .eq('id', hid);
    if (error) throw error;
    const result = {};
    for (const row of data) result[row.key] = row.value;
    return result;
  },

  async loadTx(hid, year) {
    if (!supabase) return lsGet(LS.TX, []);
    const { data, error } = await supabase
      .from('household_data').select('value')
      .eq('id', hid).eq('key', `tx_${year}`).maybeSingle();
    if (error) throw error;
    return Array.isArray(data?.value) ? data.value : [];
  },

  async saveTx(hid, year, value) {
    lsSave(LS.TX, value); // 로컬 캐시
    if (!supabase) return;
    const { error } = await supabase.from('household_data')
      .upsert({ id:hid, key:`tx_${year}`, value, updated_at: new Date().toISOString() });
    if (error) throw error;
  },

  async save(hid, key, value) {
    if (!supabase) return;
    const { error } = await supabase.from('household_data')
      .upsert({ id:hid, key, value, updated_at: new Date().toISOString() });
    if (error) throw error;
  },

  subscribe(hid, onUpdate) {
    if (!supabase) return null;
    const name = `realtime:household:${hid}`;
    if (this._channels.has(name)) return this._channels.get(name);
    const ch = supabase.channel(name)
      .on('postgres_changes',
        { event:'*', schema:'public', table:'household_data', filter:`id=eq.${hid}` },
        (payload) => {
          if (payload.eventType === 'DELETE') {
            if (payload.old?.key) onUpdate(payload.old.key, []);
          } else {
            onUpdate(payload.new.key, payload.new.value);
          }
        })
      .subscribe();
    this._channels.set(name, ch);
    return ch;
  },

  unsubscribe(hid) {
    const name = `realtime:household:${hid}`;
    const ch = this._channels.get(name);
    if (ch) { supabase?.removeChannel(ch); this._channels.delete(name); }
  },

  async loadPendingSos(hid) {
    if (!supabase) return [];
    const { data, error } = await supabase.from('sos_requests')
      .select('*').eq('household_id', hid).eq('status','pending');
    if (error) return [];
    return data || [];
  },

  async createSos(hid, requester, amount, reason, repayPlan) {
    if (!supabase) return;
    const { error } = await supabase.from('sos_requests')
      .insert({ household_id:hid, requester, amount, reason, repay_plan:repayPlan, status:'pending' });
    if (error) throw error;
  },

  async resolveSos(id, status) {
    if (!supabase) return;
    const { error } = await supabase.from('sos_requests')
      .update({ status, resolved_at: new Date().toISOString() }).eq('id', id);
    if (error) throw error;
  },
};
```

## 4. 오프라인 큐 (src/utils/offlineQueue.js → 인라인)

```js
// ─── OFFLINE QUEUE ──────────────────────────────────────────────────────────
const offlineQueue = {
  enqueue(key, value) {
    const q = lsGet(LS.OFFLINE_Q, []);
    const filtered = q.filter(item => item.key !== key);
    filtered.push({ key, value, ts: Date.now() });
    lsSave(LS.OFFLINE_Q, filtered);
  },
  hasQueued() { return (lsGet(LS.OFFLINE_Q, [])).length > 0; },
  async flush(hid) {
    if (!hid || !supabase) return 0;
    const q = lsGet(LS.OFFLINE_Q, []);
    if (!q.length) return 0;
    let ok = 0;
    const remaining = [];
    for (const item of q) {
      try {
        if (item.key.startsWith('tx_')) {
          await db.saveTx(hid, Number(item.key.slice(3)), item.value);
        } else {
          await db.save(hid, item.key, item.value);
        }
        ok++;
      } catch { remaining.push(item); }
    }
    lsSave(LS.OFFLINE_Q, remaining);
    return ok;
  },
  clear() { lsSet(LS.OFFLINE_Q, '[]'); },
};
```

function App() {
  // ── 앱 라이프사이클 상태 ──
  const [setupDone, setSetupDone] = React.useState(false);
  const [householdId, setHouseholdId] = React.useState(() => ls(LS.HOUSEHOLD_ID) || '');
  const [myRole, setMyRole] = React.useState(() => ls(LS.MY_ROLE) || 'husband');
  const [view, setView] = React.useState(() => ls('smtm_view') || 'home');
  const [sheetKey, setSheetKey] = React.useState(0);
  const [sheetOpen, setSheetOpen] = React.useState(false);
  const [sheetVis, setSheetVis] = React.useState(false);
  const [syncStatus, setSyncStatus] = React.useState('ok');

  // ── 공유 데이터 상태 ──

  const [fixed,     setFixedRaw] = React.useState(lsGet(LS.FIXED, []));
  const [cards,     setCardsRaw] = React.useState(lsGet(LS.CARDS, []));
  const [budget,    setBudget]   = React.useState(lsGet(LS.BUDGET, 2000000));
  const [names,     setNamesRaw] = React.useState(lsGet(LS.NAMES, {husband:'우진', wife:'지연'}));
  const [allowance, setAllowanceRaw] = React.useState(lsGet(LS.ALLOWANCE, {husband:300000, wife:300000}));
  const [notes,     setNotesRaw] = React.useState(lsGet(LS.NOTES, {}));
  const [sosRequests, setSosRequests] = React.useState([]);
  const inviteCode = React.useMemo(() => loadOrGenInviteCode(), []);

  // ── 개인 상태 ──
  const [pin, setPin] = React.useState(() => ls(LS.PIN) || '1234');

  // ── tx CRUD ──
  const addTransaction = React.useCallback((newTx) => {
    const item = { ...newTx, id: genId(), date: newTx.date || toDateKey() };
    setTxRaw(prev => {
      const next = [item, ...prev];
      lsSave(LS.TX, next);
      if (householdId) {
        const year = new Date().getFullYear();
        if (navigator.onLine) {
          db.saveTx(householdId, year, next).catch(() => offlineQueue.enqueue(`tx_${year}`, next));
        } else {
          offlineQueue.enqueue(`tx_${year}`, next);
        }
      }
      return next;
    });
  }, [householdId]);

  const deleteTransaction = React.useCallback((id) => {
    setTxRaw(prev => {
      const next = prev.filter(t => t.id !== id);
      lsSave(LS.TX, next);
      if (householdId && navigator.onLine) {
        const year = new Date().getFullYear();
        db.saveTx(householdId, year, next).catch(console.error);
      }
      return next;
    });
  }, [householdId]);

  // ── 공유 상태 업데이트 핸들러 (Supabase Realtime 수신) ──
  const updateSharedState = React.useCallback((key, value) => {
    if (key.startsWith('tx_')) { setTxRaw(value); lsSave(LS.TX, value); return; }
    switch(key) {
      case 'fixed':    setFixedRaw(value);    lsSave(LS.FIXED, value);    break;
      case 'cards':    setCardsRaw(value);    lsSave(LS.CARDS, value);    break;
      case 'budget':   setBudget(value);      lsSet(LS.BUDGET, value);    break;
      case 'names':    setNamesRaw(value);    lsSave(LS.NAMES, value);    break;
      case 'allowance':setAllowanceRaw(value);lsSave(LS.ALLOWANCE, value);break;
      case 'notes':    setNotesRaw(value);    lsSave(LS.NOTES, value);    break;
    }
  }, []);

  // ── Supabase 초기 로드 ──
  React.useEffect(() => {
    if (!householdId) return;
    setSetupDone(true);
    setSyncStatus('syncing');
    const year = new Date().getFullYear();
    db.loadTx(householdId, year)
      .then(data => { setTxRaw(data); lsSave(LS.TX, data); setSyncStatus('ok'); })
      .catch(() => setSyncStatus('error'));
    db.loadPendingSos(householdId).then(setSosRequests).catch(() => {});
    // 기타 데이터 로드

      if (all.fixed)    { setFixedRaw(all.fixed);    lsSave(LS.FIXED, all.fixed); }
      if (all.cards)    { setCardsRaw(all.cards);    lsSave(LS.CARDS, all.cards); }
      if (all.budget != null) { setBudget(all.budget); lsSet(LS.BUDGET, all.budget); }
      if (all.names)    { setNamesRaw(all.names);    lsSave(LS.NAMES, all.names); }
      if (all.allowance){ setAllowanceRaw(all.allowance); lsSave(LS.ALLOWANCE, all.allowance); }
    }).catch(console.error);
  }, [householdId]);

  // ── Supabase Realtime 구독 ──
  React.useEffect(() => {
    if (!householdId) return;
    db.subscribe(householdId, updateSharedState);
    return () => db.unsubscribe(householdId);
  }, [householdId, updateSharedState]);

  // ── 오프라인 큐 플러시 ──
  React.useEffect(() => {

    window.addEventListener('online', handler);

  }, [householdId]);

  // ── 입력 시트 열기 (sheetKey로 폼 리셋) ──

    setSheetKey(k => k+1);

    addTransaction, deleteTransaction,
    setFixed:    (v) => { setFixedRaw(v);    lsSave(LS.FIXED, v);    if(householdId) db.save(householdId,'fixed',v).catch(console.error); },
    setCards:    (v) => { setCardsRaw(v);    lsSave(LS.CARDS, v);    if(householdId) db.save(householdId,'cards',v).catch(console.error); },
    setBudget:   (v) => { setBudget(v);      lsSet(LS.BUDGET, v);    if(householdId) db.save(householdId,'budget',v).catch(console.error); },
    setNames:    (v) => { setNamesRaw(v);    lsSave(LS.NAMES, v);    if(householdId) db.save(householdId,'names',v).catch(console.error); },
    setAllowance:(v) => { setAllowanceRaw(v);lsSave(LS.ALLOWANCE, v);if(householdId) db.save(householdId,'allowance',v).catch(console.error); },
    setNotes:    (v) => { setNotesRaw(v);    lsSave(LS.NOTES, v);    if(householdId) db.save(householdId,'notes',v).catch(console.error); },

function OnboardingView({ onDone }) {

// 이미지 최적화 (src/utils/ocr.js optimizeImage 이식)
async function optimizeImage(file, mode='single') {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => {
      const MAX = mode === 'bulk' ? 1200 : 800;
      let { width:w, height:h } = img;
      if (w > MAX) { h = h*(MAX/w); w = MAX; }
      if (h > MAX) { w = w*(MAX/h); h = MAX; }
      const canvas = document.createElement('canvas');
      canvas.width = w; canvas.height = h;
      const ctx = canvas.getContext('2d');
      if (mode === 'single') {
        ctx.filter = 'grayscale(1) contrast(1.4) brightness(1.05)';
        ctx.drawImage(img, 0, 0, w, h);
        const id = ctx.getImageData(0,0,w,h);
        for (let i=0;i<id.data.length;i+=4) {
          const bw = id.data[i] > 160 ? 255 : 0;
          id.data[i] = id.data[i+1] = id.data[i+2] = bw;
        }
        ctx.putImageData(id, 0, 0);
        resolve({ base64: canvas.toDataURL('image/png').split(',')[1], mediaType:'image/png' });
      } else {
        ctx.drawImage(img, 0, 0, w, h);
        const webp = canvas.toDataURL('image/webp', 0.8);
        const isWebP = webp.startsWith('data:image/webp');
        const dataUrl = isWebP ? webp : canvas.toDataURL('image/jpeg', 0.8);
        const mediaType = isWebP ? 'image/webp' : 'image/jpeg';
        resolve({ base64: dataUrl.split(',')[1], mediaType });
      }
    };
    img.onerror = reject;
    img.src = URL.createObjectURL(file);
  });
}
